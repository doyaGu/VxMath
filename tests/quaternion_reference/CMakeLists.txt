cmake_minimum_required(VERSION 3.20)
project(QuaternionOracle LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
    message(FATAL_ERROR "The reference runtime requires MSVC Win32")
endif()
add_executable(quaternion_oracle main.cpp)
target_compile_features(quaternion_oracle PRIVATE cxx_std_17)
add_executable(quaternion_conversion_oracle conversions.cpp)
target_compile_features(quaternion_conversion_oracle PRIVATE cxx_std_17)
add_executable(matrix_decomposition_oracle decomposition.cpp)
target_compile_features(matrix_decomposition_oracle PRIVATE cxx_std_17)
