project(VxMathNumericalTests LANGUAGES CXX)
enable_testing()

get_filename_component(vx_root "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)

add_executable(ScalarNativeTest ScalarNativeTest.cpp)
target_include_directories(ScalarNativeTest PRIVATE "${vx_root}/include")
target_compile_features(ScalarNativeTest PRIVATE cxx_std_11)
target_compile_definitions(ScalarNativeTest PRIVATE VX_LIB VXMATH_ENABLE_X87_EMULATION=0)
add_test(NAME ScalarNativeTest COMMAND ScalarNativeTest)

add_subdirectory("${vx_root}/deps/softfloat" softfloat)

set(vx_environment_sources "${vx_root}/src/VxFPUControlWord.cpp")
if (MSVC AND CMAKE_CXX_COMPILER_ARCHITECTURE_ID STREQUAL "x64")
    enable_language(ASM_MASM)
    list(APPEND vx_environment_sources "${vx_root}/src/VxFPUControlWordWin64.asm")
endif ()

function(vx_add_numerical_library target emulate scalar)
    add_library(${target} STATIC "${vx_root}/src/VxVectorMath.cpp" ${vx_environment_sources})
    if (emulate)
        target_sources(${target} PRIVATE "${vx_root}/src/VxFloat80.cpp" "${vx_root}/src/VxFloat80Math.cpp")
        target_link_libraries(${target} PRIVATE vx_softfloat)
    endif ()
    target_include_directories(${target} PUBLIC "${vx_root}/include")
    target_compile_features(${target} PUBLIC cxx_std_17)
    target_compile_definitions(${target} PUBLIC VX_LIB VXMATH_ENABLE_X87_EMULATION=$<BOOL:${emulate}>)
    if (scalar)
        target_compile_definitions(${target} PUBLIC VX_SIMD_NONE=1)
    endif ()
    vx_enable_float80_ipo(${target})
    if (MSVC)
        target_compile_options(${target} PUBLIC /fp:precise)
    else ()
        target_compile_options(${target} PUBLIC -fno-strict-aliasing -ffp-contract=off)
    endif ()
endfunction()
